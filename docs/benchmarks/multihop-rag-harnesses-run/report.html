Configuration that ran
# The agent harnesses of the published benchmark (docs/benchmarks/multihop-rag.md), on a stratified 150 of its questions: on this corpus
# they cost 3-4x what they did on the demo, more than their share of the budget for 300. Three systems of configs/all_solutions.yaml run
# next to them, configured identically, so the harnesses are compared with the built-ins on the same questions (the disk cache answers
# those three for the 149 questions the subsets share).
#
#   ragbench datasets pull multihop-rag --subset 150 --seed 42 --output data/multihop-rag-150
#   ragbench run --config configs/all_solutions_harnesses.yaml --mock
#
# A live run starts `claude`, `codex` and `opencode` (each with its own key or login); `ragbench estimate` cannot price them.
run:
  name: all_solutions_harnesses_multihop
  output_dir: results

dataset:
  documents_path: data/multihop-rag-150/docs
  questions_path: data/multihop-rag-150/questions.jsonl
  qrels_path: data/multihop-rag-150/qrels.jsonl

systems:
  - type: no_retrieval
    name: no_retrieval_floor
    models:
      generator: gpt-6-luna

  - type: hybrid_rerank
    name: hybrid_rerank_default
    chunker:
      type: word
      chunk_size: 500
      chunk_overlap: 80
    retrieval:
      vector_store: numpy
      bm25_top_k: 20
      vector_top_k: 20
      candidate_top_k: 30
      final_top_k: 5
      rrf_k: 60
      reranker: local_relevance
      multi_query: true
      max_query_variants: 4
    models:
      embedding: text-embedding-3-small
      generator: gpt-6-luna

  - type: agent_search
    name: agent_search_tools
    chunker:
      type: word
      chunk_size: 250
      chunk_overlap: 40
    retrieval:
      retriever: hybrid
      vector_store: numpy
      max_steps: 5
      max_cost_usd: 0.05
      top_k: 5
    tools: [calculator, date_calc, corpus_grep]
    models:
      embedding: text-embedding-3-small
      generator: gpt-6-luna

  - type: cli_agent
    name: claude_code
    retrieval:
      preset: claude_code
      model: anthropic:claude-haiku-5-5
      max_steps: 20
      timeout_s: 300
      limits: {max_concurrent_requests: 2}

  - type: cli_agent
    name: codex
    retrieval:
      preset: codex
      model: gpt-6-luna
      max_steps: 20
      timeout_s: 300
      limits: {max_concurrent_requests: 2}

  - type: cli_agent
    name: opencode_glm
    retrieval:
      preset: opencode
      model: openrouter:z-ai/glm-5.3
      max_steps: 20
      timeout_s: 300
      limits: {max_concurrent_requests: 2}

openrouter:
  provider:
    data_collection: deny

evaluation:
  k_values: [1, 3, 5, 10]
  judge_enabled: true
  # Haiku 5.5 judges every answer, except those of its own family (claude_code), which gpt-6-luna judges: one judge per answer, never its own family.
  judge:
    panel: [anthropic:claude-haiku-5-5, gpt-6-luna]
    independent: true
    scorers: first
  trials: 1
  max_questions: null
  max_workers: 4
  max_cost_usd: 7.0   # the published run's cap, raised from 6.0 to finish (charged at standalone prices; Codex's tokens at API rates)
  stats:
    baseline: hybrid_rerank_default   # significance.csv: each harness against a built-in pipeline, on the same questions

# MultiHop-RAG is ~1.4M tokens: indexing it at full concurrency can exceed OpenAI's lowest-tier limit of 1M embedding tokens per minute.
limits:
  tokens_per_minute: 900000
